create table public.school_profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  full_name text not null default 'Student',
  role text not null default 'student' check (role in ('student', 'teacher')),
  created_at timestamptz not null default now()
);

create table public.school_posts (
  id uuid primary key default gen_random_uuid(),
  author_id uuid not null references public.school_profiles(id) on delete restrict,
  post_type text not null check (post_type in ('message', 'announcement')),
  title text not null,
  body text not null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index school_posts_created_at_idx on public.school_posts (created_at desc);

create or replace function public.create_school_profile_for_new_user()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
begin
  insert into public.school_profiles (id, full_name, role)
  values (
    new.id,
    coalesce(nullif(trim(new.raw_user_meta_data ->> 'full_name'), ''), 'Student'),
    'student'
  )
  on conflict (id) do nothing;
  return new;
end;
$$;

revoke all on function public.create_school_profile_for_new_user() from public, anon, authenticated;

create trigger on_auth_user_created_school_profile
after insert on auth.users
for each row execute function public.create_school_profile_for_new_user();

alter table public.school_profiles enable row level security;
alter table public.school_posts enable row level security;

revoke all on public.school_profiles from anon, authenticated;
grant select on public.school_profiles to authenticated;
create policy "Users read their own school profile"
on public.school_profiles for select to authenticated
using ((select auth.uid()) = id);

revoke all on public.school_posts from anon, authenticated;
grant select, insert, update, delete on public.school_posts to authenticated;
create policy "Authenticated users read school posts"
on public.school_posts for select to authenticated
using (true);
create policy "Teachers create school posts"
on public.school_posts for insert to authenticated
with check (
  author_id = (select auth.uid())
  and exists (
    select 1 from public.school_profiles p
    where p.id = (select auth.uid()) and p.role = 'teacher'
  )
);
create policy "Teachers update their own school posts"
on public.school_posts for update to authenticated
using (
  author_id = (select auth.uid())
  and exists (
    select 1 from public.school_profiles p
    where p.id = (select auth.uid()) and p.role = 'teacher'
  )
)
with check (
  author_id = (select auth.uid())
  and exists (
    select 1 from public.school_profiles p
    where p.id = (select auth.uid()) and p.role = 'teacher'
  )
);
create policy "Teachers delete their own school posts"
on public.school_posts for delete to authenticated
using (
  author_id = (select auth.uid())
  and exists (
    select 1 from public.school_profiles p
    where p.id = (select auth.uid()) and p.role = 'teacher'
  )
);
